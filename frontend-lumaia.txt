import React, { useState } from 'react';
import { View, Text, ScrollView, TouchableOpacity, StyleSheet, ActivityIndicator, Modal, Pressable, TextInput, KeyboardAvoidingView, Platform, Alert } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import DateTimePicker from '@react-native-community/datetimepicker';
import { Calendar, MapPin, Clock, Pill, Stethoscope, CheckCircle, Plus, StickyNote, X } from 'lucide-react-native';

import { useAuthStore } from '../store/useAuthStore';
import { useAppTheme } from '../store/useThemeStore';
import { obtenerAdultosVinculados } from '../services/cuidador.service';
import { obtenerMedicamentos, crearMedicamento } from '../services/medicamentos.service';
import { obtenerCitas, crearCita } from '../services/citas.service';
import { obtenerNotas, crearNota } from '../services/notas.service';

type ModalState = 'none' | 'menu' | 'medicamento' | 'cita' | 'nota';

export const CuidadorGestionScreen = () => {
  const { usuarioId } = useAuthStore();
  const { colors, typography } = useAppTheme();
  const queryClient = useQueryClient();
  
  const [selectedAdultoId, setSelectedAdultoId] = useState<string | null>(null);
  const [modalVisible, setModalVisible] = useState<ModalState>('none');

  // Estados Formulario Medicamento
  const [medNombre, setMedNombre] = useState('');
  const [medDosis, setMedDosis] = useState('');
  const [medFrecuencia, setMedFrecuencia] = useState('');
  const [medPrimeraToma, setMedPrimeraToma] = useState(new Date());
  const [showMedTimePicker, setShowMedTimePicker] = useState(false);

  // Estados Formulario Cita
  const [citaTitulo, setCitaTitulo] = useState('');
  const [citaUbicacion, setCitaUbicacion] = useState('');
  const [citaFechaObj, setCitaFechaObj] = useState(new Date());
  const [citaHoraObj, setCitaHoraObj] = useState(new Date());
  const [showCitaDatePicker, setShowCitaDatePicker] = useState(false);
  const [showCitaTimePicker, setShowCitaTimePicker] = useState(false);

  // Estados Formulario Nota
  const [notaContenido, setNotaContenido] = useState('');

  // Queries
  const { data: vinculados = [], isLoading: loadVinculados } = useQuery({
    queryKey: ['vinculadosList', usuarioId],
    queryFn: () => obtenerAdultosVinculados(usuarioId!),
    enabled: !!usuarioId,
  });

  // Autoseleccionar el primero si no hay seleccionado
  if (!selectedAdultoId && vinculados.length > 0) {
    setSelectedAdultoId(vinculados[0].adulto._id);
  }

  const { data: medicamentos = [], isLoading: loadMed } = useQuery({ queryKey: ['medicamentos', selectedAdultoId], queryFn: () => obtenerMedicamentos(selectedAdultoId!), enabled: !!selectedAdultoId });
  const { data: citas = [], isLoading: loadCitas } = useQuery({ queryKey: ['citas', selectedAdultoId], queryFn: () => obtenerCitas(selectedAdultoId!), enabled: !!selectedAdultoId });
  const { data: notas = [], isLoading: loadNotas } = useQuery({ queryKey: ['notas', selectedAdultoId], queryFn: () => obtenerNotas(selectedAdultoId!), enabled: !!selectedAdultoId });

  const isLoading = loadVinculados || loadMed || loadCitas || loadNotas;

  // Mutaciones (usando el ID del adulto seleccionado)
  const mutMed = useMutation({
    mutationFn: () => {
      const hh = medPrimeraToma.getHours().toString().padStart(2, '0');
      const mm = medPrimeraToma.getMinutes().toString().padStart(2, '0');
      return crearMedicamento({ usuario_id: selectedAdultoId!, nombre_farmaco: medNombre.trim(), dosis: medDosis.trim(), frecuencia_horas: parseInt(medFrecuencia) || 24, horarios_especificos: [`${hh}:${mm}`] });
    },
    onSuccess: () => resetAndClose('Medicamento asignado al paciente.', 'medicamentos')
  });

  const mutCita = useMutation({
    mutationFn: () => {
      const isoDate = new Date(citaFechaObj.getFullYear(), citaFechaObj.getMonth(), citaFechaObj.getDate(), citaHoraObj.getHours(), citaHoraObj.getMinutes()).toISOString();
      return crearCita({ usuario_id: selectedAdultoId!, titulo_evento: citaTitulo.trim(), fecha_hora: isoDate, ubicacion: citaUbicacion.trim() });
    },
    onSuccess: () => resetAndClose('Cita programada exitosamente.', 'citas')
  });

  const mutNota = useMutation({
    mutationFn: () => crearNota({ usuario_id: selectedAdultoId!, contenido: notaContenido.trim(), fecha_creacion: new Date().toISOString() }),
    onSuccess: () => resetAndClose('Nota agregada al expediente.', 'notas')
  });

  const resetAndClose = (mensaje: string, queryKey: string) => {
    Alert.alert('Éxito', mensaje);
    setModalVisible('none');
    setMedNombre(''); setMedDosis(''); setMedFrecuencia(''); setMedPrimeraToma(new Date());
    setCitaTitulo(''); setCitaUbicacion(''); setCitaFechaObj(new Date()); setCitaHoraObj(new Date());
    setNotaContenido('');
    queryClient.invalidateQueries({ queryKey: [queryKey, selectedAdultoId] });
  };

  const formatTimeUI = (date: Date) => `${date.getHours().toString().padStart(2, '0')}:${date.getMinutes().toString().padStart(2, '0')}`;
  const formatDateUI = (date: Date) => date.toLocaleDateString('es-MX', { weekday: 'short', day: '2-digit', month: 'short', year: 'numeric' });

  if (vinculados.length === 0) {
    return (
      <SafeAreaView style={{ flex: 1, backgroundColor: colors.background, justifyContent: 'center', alignItems: 'center', padding: 20 }}>
         <Text style={{ fontSize: typography.h2, fontWeight: 'bold', color: colors.textPrimary, textAlign: 'center' }}>Sin Pacientes</Text>
         <Text style={{ fontSize: typography.base, color: colors.textSecondary, textAlign: 'center', marginTop: 8 }}>Agrega un adulto mayor para gestionar su agenda.</Text>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={{ flex: 1, backgroundColor: colors.background }}>
      <View style={[styles.header, { backgroundColor: colors.surface, borderBottomColor: colors.border }]}>
        <Text style={[styles.headerTitle, { color: colors.textPrimary, fontSize: typography.h1 }]}>Gestión Remota</Text>
        <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={styles.filterContainer}>
          {vinculados.map((item) => (
            <TouchableOpacity key={item.adulto._id} onPress={() => setSelectedAdultoId(item.adulto._id)} activeOpacity={0.8}
              style={[styles.filterPill, { backgroundColor: selectedAdultoId === item.adulto._id ? colors.primary : colors.background, borderColor: selectedAdultoId === item.adulto._id ? colors.primary : colors.border }]}>
              <Text style={{ color: selectedAdultoId === item.adulto._id ? '#fff' : colors.textSecondary, fontSize: typography.base, fontWeight: 'bold' }}>{item.adulto.nombre}</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>
      </View>

      <ScrollView contentContainerStyle={styles.scrollContent}>
        {isLoading && (
          <View style={styles.centerContainer}>
            <ActivityIndicator size="large" color={colors.primary} />
          </View>
        )}

        {!isLoading && (
          <View style={styles.listContainer}>
            {citas.map((cita) => (
              <View key={cita._id} style={[styles.card, { backgroundColor: colors.surface, borderColor: colors.border, borderLeftColor: colors.warning }]}><View style={styles.cardHeader}><View style={[styles.iconBox, { backgroundColor: colors.warning + '15' }]}><Stethoscope color={colors.warning} size={32} /></View><View style={styles.cardInfo}><Text style={[styles.cardTitle, { color: colors.textPrimary, fontSize: typography.xl }]}>{cita.titulo_evento}</Text><View style={styles.badgeContainer}><Text style={[styles.badgeText, { color: colors.warning, fontSize: typography.sm }]}>CITA MÉDICA</Text></View></View></View><View style={[styles.cardDetails, { borderTopColor: colors.border }]}><View style={styles.detailRow}><Calendar color={colors.textSecondary} size={20} /><Text style={[styles.detailText, { color: colors.textSecondary, fontSize: typography.base }]}>{new Date(cita.fecha_hora).toLocaleDateString('es-MX', { weekday: 'long', day: 'numeric', month: 'long' })} a las {new Date(cita.fecha_hora).toLocaleTimeString('es-MX', { hour: '2-digit', minute: '2-digit' })}</Text></View></View></View>
            ))}
            {medicamentos.map((med) => (
              <View key={med._id} style={[styles.card, { backgroundColor: colors.surface, borderColor: colors.border, borderLeftColor: colors.primary }]}><View style={styles.cardHeader}><View style={[styles.iconBox, { backgroundColor: colors.primaryLight }]}><Pill color={colors.primary} size={32} /></View><View style={styles.cardInfo}><Text style={[styles.cardTitle, { color: colors.textPrimary, fontSize: typography.xl }]}>{med.nombre_farmaco}</Text><View style={styles.badgeContainerPrimary}><Text style={[styles.badgeTextPrimary, { color: colors.primary, fontSize: typography.sm }]}>MEDICAMENTO</Text></View></View></View><View style={[styles.cardDetails, { borderTopColor: colors.border }]}><View style={styles.detailRow}><Clock color={colors.textSecondary} size={20} /><Text style={[styles.detailText, { color: colors.textSecondary, fontSize: typography.base, fontWeight: 'bold' }]}>Cada {med.frecuencia_horas} hrs</Text></View></View></View>
            ))}
            {notas.map((nota) => (
              <View key={nota._id} style={[styles.card, { backgroundColor: '#fefce8', borderColor: '#fde047', borderLeftColor: '#eab308' }]}><View style={styles.cardHeader}><View style={[styles.iconBox, { backgroundColor: '#fef08a' }]}><StickyNote color="#ca8a04" size={32} /></View><View style={styles.cardInfo}><Text style={[styles.cardTitle, { color: colors.textPrimary, fontSize: typography.xl }]}>{nota.contenido}</Text></View></View></View>

            {citas.length === 0 && medicamentos.length === 0 && notas.length === 0 && (

// [MÓVIL] Ruta: src/services/cuidador.service.ts

import axios from 'axios';
import { apiClient } from '../config/api.client';
import { Usuario, Medicamento, Cita, RegistroToma } from '../types';

export interface ResumenAdultoVinculado {
  adulto: Usuario;
  medicamentos: Medicamento[];
  citas_hoy: Cita[];
  tomas_hoy: RegistroToma[];
}

const handleGraphQLError = (error: unknown) => {
  if (axios.isAxiosError(error) && error.response?.data?.errors) {
    throw new Error(error.response.data.errors[0].message);
  }
  throw new Error('Ocurrió un error inesperado al conectar con el servidor.');
};

export const vincularAdultoMayor = async (cuidador_id: string, correo_adulto: string): Promise<Usuario> => {
  const mutation = `
    mutation VincularAdultoMayor($cuidador_id: ID!, $correo_adulto: String!) {
      vincularAdultoMayor(cuidador_id: $cuidador_id, correo_adulto: $correo_adulto) {
        _id
        nombre
        correo
      }
    }
  `;
  try {
    const response = await apiClient.post('', { query: mutation, variables: { cuidador_id, correo_adulto } });
    if (response.data.errors?.length > 0) throw new Error(response.data.errors[0].message);
    return response.data.data.vincularAdultoMayor;
  } catch (error) {
    handleGraphQLError(error);
    throw error;
  }
};

export const obtenerAdultosVinculados = async (cuidador_id: string): Promise<ResumenAdultoVinculado[]> => {
  const query = `
    query ObtenerAdultosVinculados($cuidador_id: ID!) {
      obtenerAdultosVinculados(cuidador_id: $cuidador_id) {
        adulto {
          _id
          nombre
          correo
          contactos_emergencia { telefono }
        }
        medicamentos { _id nombre_farmaco frecuencia_horas }
        citas_hoy { _id titulo_evento fecha_hora ubicacion }
        tomas_hoy { _id medicamento_id }
      }
    }
  `;
  try {
    const response = await apiClient.post('', { query, variables: { cuidador_id } });
    if (response.data.errors?.length > 0) throw new Error(response.data.errors[0].message);
    return response.data.data.obtenerAdultosVinculados || [];
  } catch (error) {
    handleGraphQLError(error);
    return [];
  }
};

import axios from 'axios';

// RECUERDA: Cambiar a la IP local de tu PC donde corre NestJS
export const apiClient = axios.create({
  baseURL: '  https://rebuilt-shakily-cinch.ngrok-free.dev/graphql', 
  timeout: 45000, // <-- CORREGIDO: 45 segundos para dar tiempo a Ollama
  headers: {
    'Content-Type': 'application/json',
  },
});

apiClient.interceptors.request.use((config) => {
  const token = useAuthStore.getState().token;
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

apiClient.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.code === 'ECONNABORTED' || error.message.includes('timeout')) {
      return Promise.reject(new Error('El servidor tardó demasiado en responder. Si usas IA Local, espera un momento.'));
    }
    if (!error.response) {
      return Promise.reject(new Error('No se pudo conectar con el servidor. Verifica tu red.'));
    }
    return Promise.reject(error);
  }
);

// Extractor de Errores GraphQL
const handleGraphQLError = (error: unknown) => {
  if (axios.isAxiosError(error) && error.response?.data?.errors) {
    throw new Error(error.response.data.errors[0].message);
  }
  if (error instanceof Error) {
    throw error;
  }
  throw new Error('Ocurrió un error inesperado');
};

const mapGraphQLRoleToEnum = (gqlRole: string): Enum_Rol => {
  if (gqlRole === 'ADULTO_MAYOR') return 'Adulto Mayor';
  if (gqlRole === 'CUIDADOR') return 'Cuidador';
  return 'Adulto Mayor'; 
};

export const mapEnumToGraphQLRole = (rol: Enum_Rol): string => {
  if (rol === 'Adulto Mayor') return 'ADULTO_MAYOR';
  if (rol === 'Cuidador') return 'CUIDADOR';
  return 'ADULTO_MAYOR';
};

export const loginUsuario = async (correo: string, password: string): Promise<LoginResponse> => {
  const query = `
    mutation Login($correo: String!, $password: String!) {
      login(correo: $correo, password: $password) {
        access_token
        usuario_id
        rol
      }
    }
  `;

  try {
    const response = await apiClient.post('', {
      query,
      variables: { correo, password },
    });

    if (response.data.errors?.length > 0) {
      throw new Error(response.data.errors[0].message);
    }

    const data = response.data.data.login;
    return {
      access_token: data.access_token,
      usuario_id: data.usuario_id,
      rol: mapGraphQLRoleToEnum(data.rol),
    };
  } catch (error) {
    handleGraphQLError(error);
    throw error;
  }
};

import { create } from 'zustand';

type FontSizeSize = 'normal' | 'large' | 'xlarge';
type ContrastMode = 'normal' | 'high';

interface ThemeState {
  fontSize: FontSizeSize;
  contrast: ContrastMode;
  setFontSize: (size: FontSizeSize) => void;
  setContrast: (mode: ContrastMode) => void;
}

export const useThemeStore = create<ThemeState>((set) => ({
  fontSize: 'normal',
  contrast: 'normal',
  setFontSize: (size) => set({ fontSize: size }),
  setContrast: (mode) => set({ contrast: mode }),
}));

// Hook generador de Tema para usar en los componentes
export const useAppTheme = () => {
  const { fontSize, contrast } = useThemeStore();

  // Factor de escala de fuente
  const fontScale = fontSize === 'normal' ? 1 : fontSize === 'large' ? 1.2 : 1.4;

  // Paleta de Colores Dinámica
  const colors = {
    background: contrast === 'high' ? '#ffffff' : '#f8fafc', // slate-50
    surface: contrast === 'high' ? '#ffffff' : '#ffffff',
    textPrimary: contrast === 'high' ? '#000000' : '#0f172a', // slate-900
    textSecondary: contrast === 'high' ? '#333333' : '#475569', // slate-600
    border: contrast === 'high' ? '#000000' : '#e2e8f0', // slate-200
    primary: contrast === 'high' ? '#0000ee' : '#2563eb', // blue-600
    primaryLight: contrast === 'high' ? '#e6e6ff' : '#eff6ff', // blue-50
    success: contrast === 'high' ? '#006600' : '#059669', // emerald-600
    successLight: contrast === 'high' ? '#e6ffe6' : '#ecfdf5', // emerald-50
    danger: contrast === 'high' ? '#cc0000' : '#e11d48', // rose-600
    warning: contrast === 'high' ? '#660066' : '#4f46e5', // indigo-600
  };

  // Tamaños de texto dinámicos
  const typography = {
    xs: 12 * fontScale,
    sm: 14 * fontScale,
    base: 16 * fontScale,
    lg: 18 * fontScale,
    xl: 20 * fontScale,
    h2: 24 * fontScale,
    h1: 32 * fontScale,
  };

  return { colors, typography };
};

    { id: 'init', emisor: 'bot', texto: `¡Hola! Soy Luma. ¿En qué te puedo ayudar hoy? 😊` }
  ]);
  
  const [inputText, setInputText] = useState('');
  const [modeloIA, setModeloIA] = useState<ModeloIA>('Luma Cloud');
  const [modalModeloVisible, setModalModeloVisible] = useState(false);
  
  // ESTADOS DE AUDIO
  const [recording, setRecording] = useState<Audio.Recording | null>(null);
  const [isListening, setIsListening] = useState(false);

  const reproducirAudio = (texto: string) => {
    Speech.stop(); 
    Speech.speak(texto, { language: 'es-MX', rate: 0.9 });
  };

  const mutation = useMutation({
    mutationFn: (pregunta_id: string) => consultarAsistente({ usuario_id: usuarioId!, pregunta_id, modelo: modeloIA }),
    onSuccess: (respuesta) => {
      setHistorial(prev => [...prev, { id: Date.now().toString(), emisor: 'bot', texto: respuesta }]);
      reproducirAudio(respuesta);
    },
    onError: (error) => {
      const msjError = error instanceof Error ? error.message : 'Problema al procesar tu consulta.';
      setHistorial(prev => [...prev, { id: Date.now().toString(), emisor: 'bot', texto: `Error: ${msjError}` }]);
    }
  });

  const hacerPregunta = (textoPregunta: string, idOculto?: string) => {
    if (mutation.isPending) return;
    const promptFinal = idOculto ? idOculto : textoPregunta;
    setHistorial(prev => [...prev, { id: Date.now().toString(), emisor: 'user', texto: textoPregunta }]);
    mutation.mutate(promptFinal);
    setInputText('');
  };

  // --- LÓGICA DE RECONOCIMIENTO DE VOZ ---
  const startRecordingMobile = async () => {
    try {
      const permission = await Audio.requestPermissionsAsync();
      if (permission.status === 'granted') {
        await Audio.setAudioModeAsync({ allowsRecordingIOS: true, playsInSilentModeIOS: true });
        const { recording } = await Audio.Recording.createAsync(Audio.RecordingOptionsPresets.HIGH_QUALITY);
        setRecording(recording);
        setIsListening(true);
      } else {
        Alert.alert('Permiso denegado', 'Necesitamos acceso al micrófono para que Luma te escuche.');
      }
    } catch (err) {
      console.error('Failed to start recording', err);
    }
  };

  const stopRecordingMobile = async () => {
    if (!recording) return;
    setIsListening(false);
    await recording.stopAndUnloadAsync();
    const uri = recording.getURI();
    setRecording(null);
    
    if (uri) {
       // Leemos el archivo en Base64
       const base64Audio = await FileSystem.readAsStringAsync(uri, { encoding: FileSystem.EncodingType.Base64 });
       // AQUÍ: Llamaríamos a la mutación para que Whisper en NestJS transcriba el Base64 a texto.
       // Por ahora, simulamos que el texto llega y lo ponemos en el input:
       Alert.alert('Audio Capturado', 'Audio convertido a Base64 listo para enviar al backend (Transcriptor Whisper en desarrollo).');
    }
  };

  const startWebSpeech = () => {
    const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
    if (!SpeechRecognition) {
        Alert.alert('Error', 'Tu navegador no soporta dictado por voz de forma nativa.');
        return;
    }
    setIsListening(true);
    const recognition = new SpeechRecognition();
    recognition.lang = 'es-MX';
    recognition.continuous = false;
    recognition.interimResults = false;

    recognition.onresult = (event: any) => {
        const text = event.results[0][0].transcript;
        setInputText(prev => (prev + ' ' + text).trim());
        setIsListening(false);
    };
    recognition.onerror = () => setIsListening(false);
    recognition.onend = () => setIsListening(false);
    recognition.start();
  };

  const handleMicPress = () => {
    if (isListening) {
      Platform.OS === 'web' ? setIsListening(false) : stopRecordingMobile();
    } else {
      Platform.OS === 'web' ? startWebSpeech() : startRecordingMobile();
    }
  };

  return (

      <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : 'height'} style={{ flex: 1 }}>

          {historial.map((msg) => (

                      <Text style={[styles.menuBtnText, { color: colors.textPrimary, fontSize: typography.xl }]}>Medicamento</Text>
                    </TouchableOpacity>
                    <TouchableOpacity style={[styles.menuBtn, { borderColor: colors.border }]} onPress={() => setModalVisible('cita')}>
                      <View style={[styles.menuIconBox, { backgroundColor: colors.warning + '15' }]}><Calendar color={colors.warning} size={32} /></View>
                      <Text style={[styles.menuBtnText, { color: colors.textPrimary, fontSize: typography.xl }]}>Cita Médica</Text>
                    </TouchableOpacity>
                    <TouchableOpacity style={[styles.menuBtn, { borderColor: colors.border }]} onPress={() => setModalVisible('nota')}>
                      <View style={[styles.menuIconBox, { backgroundColor: '#fef08a' }]}><StickyNote color="#ca8a04" size={32} /></View>
                      <Text style={[styles.menuBtnText, { color: colors.textPrimary, fontSize: typography.xl }]}>Nota Personal</Text>
                    </TouchableOpacity>
                  </View>
                )}

                {modalVisible === 'medicamento' && (
                  <View style={{ paddingBottom: 20 }}>
                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg }]}>Nombre del Fármaco</Text>
                    <TextInput style={[styles.input, { borderColor: colors.border, color: colors.textPrimary, fontSize: typography.lg }]} placeholder="Ej. Losartán" placeholderTextColor={colors.textSecondary} value={medNombre} onChangeText={setMedNombre} />
                    
                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg, marginTop: 16 }]}>Dosis Recomendada</Text>
                    <TextInput style={[styles.input, { borderColor: colors.border, color: colors.textPrimary, fontSize: typography.lg }]} placeholder="Ej. 1 Pastilla" placeholderTextColor={colors.textSecondary} value={medDosis} onChangeText={setMedDosis} />
                    
                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg, marginTop: 16 }]}>Frecuencia (Cada cuántas horas)</Text>
                    <TextInput style={[styles.input, { borderColor: colors.border, color: colors.textPrimary, fontSize: typography.lg }]} placeholder="Ej. 8" keyboardType="numeric" placeholderTextColor={colors.textSecondary} value={medFrecuencia} onChangeText={setMedFrecuencia} />
                    
                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg, marginTop: 16 }]}>¿A qué hora es la primera toma?</Text>
                    <TouchableOpacity style={[styles.inputBtn, { borderColor: colors.border, backgroundColor: colors.background }]} onPress={() => setShowMedTimePicker(true)}>
                      <Clock color={colors.primary} size={24} style={{ marginRight: 12 }} />
                      <Text style={{ color: colors.textPrimary, fontSize: typography.lg, fontWeight: 'bold' }}>{formatTimeUI(medPrimeraToma)}</Text>
                    </TouchableOpacity>
                    {showMedTimePicker && (<DateTimePicker value={medPrimeraToma} mode="time" display="default" onChange={onChangeMedTime} />)}

                    <TouchableOpacity style={[styles.submitBtn, { backgroundColor: colors.primary, opacity: mutMed.isPending ? 0.7 : 1 }]} onPress={() => mutMed.mutate()} disabled={mutMed.isPending || !medNombre || !medDosis}>
                      {mutMed.isPending ? <ActivityIndicator color="#fff" /> : <Text style={{ color: '#fff', fontSize: typography.h2, fontWeight: '900' }}>Guardar</Text>}
                    </TouchableOpacity>
                  </View>
                )}

                {modalVisible === 'cita' && (
                  <View style={{ paddingBottom: 20 }}>
                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg }]}>Motivo o Especialidad</Text>
                    <TextInput style={[styles.input, { borderColor: colors.border, color: colors.textPrimary, fontSize: typography.lg }]} placeholder="Ej. Cardiólogo" placeholderTextColor={colors.textSecondary} value={citaTitulo} onChangeText={setCitaTitulo} />
                    
                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg, marginTop: 16 }]}>Día de la cita</Text>
                    <TouchableOpacity style={[styles.inputBtn, { borderColor: colors.border, backgroundColor: colors.background }]} onPress={() => setShowCitaDatePicker(true)}>
                      <Calendar color={colors.warning} size={24} style={{ marginRight: 12 }} />
                      <Text style={{ color: colors.textPrimary, fontSize: typography.lg, fontWeight: 'bold', textTransform: 'capitalize' }}>{formatDateUI(citaFechaObj)}</Text>
                    </TouchableOpacity>
                    {showCitaDatePicker && (<DateTimePicker value={citaFechaObj} mode="date" display="default" minimumDate={new Date()} onChange={onChangeCitaDate} />)}

                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg, marginTop: 16 }]}>Hora</Text>
                    <TouchableOpacity style={[styles.inputBtn, { borderColor: colors.border, backgroundColor: colors.background }]} onPress={() => setShowCitaTimePicker(true)}>
                      <Clock color={colors.warning} size={24} style={{ marginRight: 12 }} />
                      <Text style={{ color: colors.textPrimary, fontSize: typography.lg, fontWeight: 'bold' }}>{formatTimeUI(citaHoraObj)}</Text>
                    </TouchableOpacity>
                    {showCitaTimePicker && (<DateTimePicker value={citaHoraObj} mode="time" display="default" onChange={onChangeCitaTime} />)}

                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg, marginTop: 16 }]}>Ubicación (Opcional)</Text>
                    <TextInput style={[styles.input, { borderColor: colors.border, color: colors.textPrimary, fontSize: typography.lg }]} placeholder="Ej. Clínica Centro" placeholderTextColor={colors.textSecondary} value={citaUbicacion} onChangeText={setCitaUbicacion} />
                    
                    <TouchableOpacity style={[styles.submitBtn, { backgroundColor: colors.primary, opacity: mutCita.isPending ? 0.7 : 1 }]} onPress={() => mutCita.mutate()} disabled={mutCita.isPending || !citaTitulo}>
                      {mutCita.isPending ? <ActivityIndicator color="#fff" /> : <Text style={{ color: '#fff', fontSize: typography.h2, fontWeight: '900' }}>Programar Cita</Text>}
                    </TouchableOpacity>
                  </View>
                )}

                {modalVisible === 'nota' && (
                  <View style={{ paddingBottom: 20 }}>
                    <Text style={[styles.label, { color: colors.textPrimary, fontSize: typography.lg }]}>¿Qué deseas recordar?</Text>
                    <TextInput style={[styles.input, { borderColor: colors.border, color: colors.textPrimary, fontSize: typography.lg, minHeight: 120, textAlignVertical: 'top', paddingTop: 16 }]} placeholder="Escribe tu apunte aquí..." placeholderTextColor={colors.textSecondary} multiline value={notaContenido} onChangeText={setNotaContenido} />
                    
                    <TouchableOpacity style={[styles.submitBtn, { backgroundColor: colors.primary, opacity: mutNota.isPending ? 0.7 : 1 }]} onPress={() => mutNota.mutate()} disabled={mutNota.isPending || !notaContenido}>
                      {mutNota.isPending ? <ActivityIndicator color="#fff" /> : <Text style={{ color: '#fff', fontSize: typography.h2, fontWeight: '900' }}>Guardar Apunte</Text>}
                    </TouchableOpacity>
                  </View>
                )}

              </ScrollView>
            </View>
          </View>
        </KeyboardAvoidingView>
      </Modal>
    </SafeAreaView>
  );
};
